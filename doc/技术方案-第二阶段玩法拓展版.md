#### 3.1.2 收益计算逻辑

```typescript
// services/tradingService.ts

/**
 * 跑商收益计算
 * 基础收益 × (1 + 商人加成) × (1 - 护院加成) × 随机浮动 - 风险减值
 */
function calcTradeProfit(
  route: TradeRoute,
  members: FamilyMember[],
  servants: Servant[]
): number {
  // 1. 基础收益浮动
  const baseProfit = route.baseProfit;
  const floatFactor = 1 + (Math.random() * 2 - 1) * 0.3; // ±30%

  // 2. 商人职业加成（每个商人提升 15%）
  const merchantCount = members.filter(m => m.profession === 'merchant').length;
  const merchantBonus = 1 + merchantCount * 0.15;

  // 3. 护院家丁降低亏损概率
  const guardLevel = servants.find(s => s.type === 'guard')?.level || 0;
  const lossReduction = guardLevel * 0.05; // 每级降低 5%

  // 4. 判定是否亏损
  const adjustedLossChance = Math.max(0, route.lossChance - lossReduction);
  const isLoss = Math.random() < adjustedLossChance;

  if (isLoss) {
    return -Math.floor(baseProfit * floatFactor * merchantBonus * 0.5);
  }

  return Math.floor(baseProfit * floatFactor * merchantBonus);
}

/**
 * 跑商冷却检查
 */
function canStartMission(state: TradingState): boolean {
  return state.dailyMissionCount < state.maxDailyMissions;
}
```

#### 3.1.3 跑商流程

```
1. 选择商路 → 展示收益/风险预览
2. 选择派遣成员（拖拽或点击选择）
```

```typescript
// services/professionService.ts

/**
 * 成员转职
 * 条件：成年状态、消耗银两
 */
function changeProfession(
  member: FamilyMember,
  newProfession: Profession,
  silver: number
): { success: boolean; cost: number; message: string } {
  if (member.lifeStage !== 'adult') {
    return { success: false, cost: 0, message: '仅成年成员可转职' };
  }

  const config = PROFESSION_CONFIG[newProfession];
  const cost = config.upgradeCost(1); // 转职费用 = 第1级升级费

  if (silver < cost) {
    return { success: false, cost: 0, message: '银两不足' };
  }

  member.profession = newProfession;
  member.professionLevel = 1;
  updateMemberLifeStage(member); // 重新计算收益

  return { success: true, cost, message: '转职成功' };
}
```

```typescript
// services/servantService.ts

function calcUpgradeCost(servant: Servant): number {
  // 指数增长：基础1000 × 1.5^(等级-1)
  return Math.floor(1000 * Math.pow(1.5, servant.level - 1));
}

function upgradeServant(servant: Servant, silver: number): boolean {
  if (servant.level >= servant.maxLevel) return false;
  const cost = calcUpgradeCost(servant);
  if (silver < cost) return false;

  servant.level += 1;
  servant.upgradeCost = calcUpgradeCost(servant);
  return true;
}
```

```typescript
// services/treeService.ts

/**
 * 升级所需经验：100 × 等级²
 */
function calcExpToNextLevel(level: number): number {
  return 100 * level * level;
}

/**
 * 浇水获得经验
 * 自己浇水 +30 经验
 * 好友浇水 +20 经验
 */
function waterTree(tree: DivineTree, isFriend: boolean): void {
  tree.experience += isFriend ? 20 : 30;

  if (!isFriend) {
    tree.waterCount += 1;
  } else {
    tree.friendWaterCount += 1;
  }

  // 检查升级
  while (tree.experience >= tree.experienceToNext) {
    tree.experience -= tree.experienceToNext;
    tree.level += 1;
    tree.experienceToNext = calcExpToNextLevel(tree.level);

    // 收益加成：每级 +2%
    tree.bonusMultiplier = 1 + tree.level * 0.02;
  }
}
```

#### 3.5.2 事件触发逻辑

```typescript
// services/eventService.ts

/**
 * 每月结算后触发随机事件判定
 * 每个游戏月有 30% 概率触发随机事件
 */
function checkRandomEvent(
  gameMonth: number,
  familyState: FamilyState
): RandomEvent | null {
  // 基础触发率 30%
  if (Math.random() > 0.3) return null;

  // 筛选满足条件的可用事件
  const availableEvents = EVENT_POOL.filter(event => {
    const c = event.conditions;
    if (!c) return true;
    if (c.minMembers && familyState.memberCount < c.minMembers) return false;
    if (c.minSilver && familyState.silver < c.minSilver) return false;
    return true;
  });

  if (availableEvents.length === 0) return null;

  // 按权重随机选择
  const totalWeight = availableEvents.reduce((sum, e) => sum + e.weight, 0);
  let random = Math.random() * totalWeight;

  for (const event of availableEvents) {
    random -= event.weight;
    if (random <= 0) return event;
  }

  return availableEvents[0];
}
```

## 四、全局 Buff 聚合计算

本阶段引入了多个全局加成来源（职业、家丁、神树），需统一聚合：

```typescript
// services/buffService.ts

interface GlobalBuffs {
  incomeMultiplier: number;     // 收益倍率
  costReduction: number;        // 支出减免
  tradingBonus: number;         // 跑商加成
  lossReduction: number;        // 亏损减免
  offlineExtendHours: number;   // 离线延长时间
  aptitudeChance: number;       // 资质提升概率
}

/**
 * 从各系统汇总全局 Buff
 */
function aggregateBuffs(
  members: FamilyMember[],
  servants: Servant[],
  tree: DivineTree
): GlobalBuffs {
  const buffs: GlobalBuffs = {
    incomeMultiplier: 1,
    costReduction: 0,
    tradingBonus: 0,
    lossReduction: 0,
    offlineExtendHours: 0,
    aptitudeChance: 0,
  };

  // 神树加成
  buffs.incomeMultiplier = tree.bonusMultiplier;

  // 家丁加成
  for (const s of servants) {
    const val = s.effect.getCurrentValue(s.level);
    switch (s.type) {
      case 'accountant': buffs.incomeMultiplier += val; break;
      case 'guard': buffs.lossReduction += val; break;
      case 'butler': buffs.offlineExtendHours += val; break;
    }
  }

  // 职业全局效果
  const doctors = members.filter(m => m.profession === 'doctor');
  if (doctors.length > 0) {
    const doctorConfig = PROFESSION_CONFIG.doctor;
    doctors.forEach(d => {
      buffs.costReduction += doctorConfig.specialEffect.value * d.professionLevel;
    });
  }

  return buffs;
}
```

## 五、性能优化要点

| 优化项 | 方案 |
|--------|------|
| 跑商定时器 | 记录 `endTime` 时间戳而非使用 `setInterval`，避免后台暂停时计时器失效 |
| 结算批量 | 批量结算时一次性更新全部成员，减少 setState 触发次数 |
| 事件队列 | 多个随机事件触发时加入队列逐一展示，避免弹窗叠加 |
| 神树动画 | 神树升级动效用 CSS transition，低端机降至静态图片切换 |
| 图片资源 | 职业图标、家丁头像使用 iconfont 或 SVG，减少图片请求 |
