```typescript
// services/adService.ts

type AdScene =
  | 'reward_offline_double' | 'reward_trade_double'
  | 'reward_daily_double' | 'reward_extend_offline'
  | 'reward_refresh_aptitude' | 'reward_free_prop';

interface AdUnitConfig {
  scene: AdScene;
  wechatAdUnitId: string;
  douyinAdUnitId: string;
  dailyLimit: number;
}

class AdManager {
  private dailyCounts: Record<AdScene, number> = { /* ... */ };
  private rewardVideoAd: any = null;
  private config: AdUnitConfig[];

  constructor(config: AdUnitConfig[]) { this.config = config; }

  /** 跨天重置计数器 */
  resetDailyIfNeeded(signInState: SignInState): void {
    const today = getTodayStr();
    if (signInState.lastSignInDate !== today) {
      for (const key of Object.keys(this.dailyCounts)) {
        this.dailyCounts[key as AdScene] = 0;
      }
    }
  }

  /** 播放激励视频 */
  async showRewardVideo(scene: AdScene): Promise<boolean> {
    const cfg = this.config.find(c => c.scene === scene);
    if (!cfg) return false;
    if (this.dailyCounts[scene] >= cfg.dailyLimit) return false;

    try {
      const adUnitId = isWeChat ? cfg.wechatAdUnitId : cfg.douyinAdUnitId;
      this.rewardVideoAd = this.createRewardVideoAd(adUnitId);
      await this.rewardVideoAd.show();
      return true;
    } catch (err) {
      console.warn('广告播放失败，后续可重试', err);
      return false;
    }
  }

  private createRewardVideoAd(adUnitId: string) {
    const api = isWeChat ? wx : tt;
    return api.createRewardedVideoAd({ adUnitId });
  }
}
```

### 3.2 支付管理层

```typescript
// services/paymentService.ts

interface PaymentConfig {
  wechatOfferId: string;
  douyinOfferId: string;
}

class PaymentManager {
  private config: PaymentConfig;

  /** 发起支付 */
  async purchase(
    productId: string,
  ): Promise<{ success: boolean; orderId?: string }> {
    // 1. 调用云函数创建订单
    const order = await cloud.callFunction({
      name: 'createOrder',
      data: { productId, platform: isWeChat ? 'wechat' : 'douyin' },
    });

    // 2. 调起平台支付
    if (isWeChat) {
      return this.wechatPay(order);
    } else {
      return this.douyinPay(order);
    }
  }

  private async wechatPay(order: any): Promise<any> {
    const result = await wx.requestPayment({
      timeStamp: order.timeStamp,
      nonceStr: order.nonceStr,
      package: order.package,
      signType: 'MD5',
      paySign: order.paySign,
    });
    return result;
  }

  private async douyinPay(order: any): Promise<any> {
    const result = await tt.pay({
      orderId: order.orderId,
      orderToken: order.orderToken,
    });
    return result;
  }

  /** 发货校验——云函数服务端验证 */
  async verifyAndDeliver(orderId: string): Promise<void> {
    await cloud.callFunction({
      name: 'verifyPayment',
      data: { orderId },
    });
  }
}
```

## 五、合规与安全

### 5.1 未成年人防沉迷

```typescript
// services/guardService.ts

class YouthGuardService {
  private isMinor = false;
  private playStartTime = 0;
  private dailyPlaySeconds = 0;
  private readonly MINOR_DAILY_LIMIT = 90 * 60;  // 1.5小时
  private readonly REST_PERIOD = 21 * 3600;       // 21:00-次日8:00

  /** 初始化：在 onLaunch 调用 */
  async init(): Promise<void> {
    if (isWeChat) {
      const res = await wx.checkIsAddedToMyMiniProgram();
      this.isMinor = false; // 微信需结合实名认证结果
    } else {
      const res = await tt.getUserInfo();
      this.isMinor = res.userInfo?.isMinor || false;
    }

    this.playStartTime = Date.now();
    this.startTick();
  }

  private startTick(): void {
    setInterval(() => {
      if (!this.isMinor) return;
      this.dailyPlaySeconds += 1;

      // 时长限制
      if (this.dailyPlaySeconds >= this.MINOR_DAILY_LIMIT) {
        this.forceLogout('今日游玩时长已达上限，请明日再来');
      }

      // 宵禁时段
      const hour = new Date().getHours();
      if (hour >= 21 || hour < 8) {
        this.forceLogout('当前为未成年人禁玩时段(21:00-次日8:00)');
      }
    }, 1000);
  }

  private forceLogout(msg: string): void {
    Taro.showModal({ title: '提示', content: msg, showCancel: false });
    // 禁止操作游戏逻辑
  }

  /** 充值拦截 */
  checkPurchaseLimit(amount: number): boolean {
    if (!this.isMinor) return true;
    // 单次≤50元，月累计≤200元
    return amount <= 5000; // 分
  }
}
```

### 6.3 性能优化

| 优化项 | 方案 | 目标 |
|--------|------|------|
| 主包体积 | 非核心模块全部进分包；图片用 CDN/云存储；Taro UI 按需引入 | < 2MB |
