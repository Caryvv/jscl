```typescript
interface InheritanceCondition {
  type: 'patriarch_age' | 'prestige' | 'silver' | 'members';
  description: string;
  required: number;
  current: number;
  satisfied: boolean;
}

function checkInheritanceConditions(state: FamilyState): InheritanceCondition[] {
  const patriarch = getPatriarch(state.members);
  const conditions: InheritanceCondition[] = [
    {
      type: 'patriarch_age',
      description: '家主年满50岁',
      required: 600, // 50年×12月
      current: patriarch?.age || 0,
      satisfied: (patriarch?.age || 0) >= 600,
    },
    {
      type: 'prestige',
      description: '家族声望达到指定值',
      required: calcRequiredPrestige(state.dynasty),
      current: state.dynasty.prestige,
      satisfied: state.dynasty.prestige >= calcRequiredPrestige(state.dynasty),
    },
    {
      type: 'adult_heirs',
      description: '有成年子嗣可选继承',
      required: 1,
      current: countAdultOffspring(state.members),
      satisfied: countAdultOffspring(state.members) >= 1,
    },
  ];

  return conditions;
}
```

#### 3.1.3 传承执行流程

```typescript
/**
 * 执行传承
 * 1. 选定继承人家主
 * 2. 保留核心资产与永久 Buff
 * 3. 重置临时状态
 * 4. 更新朝代信息
 * 5. 记录族谱
 */
function executeInheritance(
  newPatriarchId: string,
  state: GameState
): { success: boolean; result: InheritanceRecord } {
  const oldPatriarch = getPatriarch(state.family.members);
  const newPatriarch = state.family.members.find(m => m.id === newPatriarchId);

  if (!oldPatriarch || !newPatriarch) {
    return { success: false, result: null };
  }

  // 保留项
  const preservedSilver = state.resources.silver;
  const preservedLuckyCharm = state.resources.luckyCharm;
  const preservedGoldBar = state.resources.goldBar;
  const preservedTree = state.tree;                   // 神树等级保留
  const preservedCollections = state.collections;     // 传家藏品保留
  const preservedBuffs = aggregateGlobalBuffs(state); // 全局 Buff 保留
  const preservedPrestige = state.dynasty.prestige + 100; // 传承增加声望

  // 变更家主身份
  oldPatriarch.role = 'collateral';
  newPatriarch.role = 'patriarch';

  // 更新后代数量
  state.dynasty.generationCount += 1;

  // 检查是否朝代更迭（每3-5代触发）
  let needsDynastyChange = false;
  if (state.dynasty.generationCount >= 5) {
    needsDynastyChange = true;
  } else if (state.dynasty.generationCount >= 3 && Math.random() < 0.3) {
    needsDynastyChange = true;
  }

  if (needsDynastyChange) {
    triggerDynastyChange(state);
  }

  // 记录传承
  const record: InheritanceRecord = {
    id: generateId(),
    dynastyId: state.dynasty.id,
    generationIndex: state.dynasty.generationCount,
    oldPatriarchId: oldPatriarch.id,
    newPatriarchId: newPatriarch.id,
    inheritedAssets: {
      silver: preservedSilver,
      luckyCharm: preservedLuckyCharm,
      goldBar: preservedGoldBar,
    },
    preservedBuffs,
    timestamp: Date.now(),
    gameMonth: state.system.gameMonth,
  };

  state.inheritanceHistory.push(record);

  return { success: true, result: record };
}
```

```typescript
function generateEpitaph(entry: GenealogyEntry): string {
  const parts: string[] = [];

  parts.push(`${entry.name}，`);

  // 身份描述
  if (entry.role === 'patriarch') {
    parts.push(`曾任第${entry.dynastyId}代家主，`);
  }

  // 职业描述
  if (entry.finalProfession !== 'none') {
    parts.push(`终职${PROFESSION_CONFIG[entry.finalProfession].name}${entry.finalProfessionLevel}级，`);
  }

  // 贡献描述
  if (entry.lifetimeIncome > 100000) {
    parts.push(`一生为家族创收十万余两，`);
  } else if (entry.lifetimeIncome > 10000) {
    parts.push(`一生为家族创收万余两，`);
  }

  // 后代
  if (entry.lifetimeChildren > 0) {
    parts.push(`育有${entry.lifetimeChildren}名子女。`);
  }

  if (entry.achievements.length > 0) {
    parts.push(`曾获「${entry.achievements.join('」「')}」等荣誉。`);
  }

  return parts.join('');
```

```typescript
// constants/achievement.ts

type AchievementCategory = 'wealth' | 'family' | 'trading' | 'inheritance' | 'collection' | 'special';

interface Achievement {
  id: string;
  name: string;
  description: string;
  category: AchievementCategory;
  icon: string;
  condition: AchievementCondition;
  rewards: { type: 'silver' | 'luckyCharm' | 'goldBar' | 'heirloom'; amount: number; itemId?: string }[];
  isHidden: boolean;          // 隐藏成就（不显示条件）
  sortOrder: number;
}

type AchievementCondition =
  | { type: 'silver_reach'; target: number }
  | { type: 'member_count'; target: number }
  | { type: 'trade_count'; target: number }
  | { type: 'inheritance_count'; target: number }
  | { type: 'collection_count'; target: number }
  | { type: 'tree_level'; target: number }
  | { type: 'dynasty_reach'; target: number }
  | { type: 'aptitude_genius_count'; target: number };

// 成就监听器模式
class AchievementTracker {
  private listeners: Map<string, (state: GameState) => number> = new Map();

  register(conditionType: string, getter: (state: GameState) => number) {
    this.listeners.set(conditionType, getter);
  }

  checkAll(state: GameState, unlockedIds: string[]): Achievement[] {
    const newlyUnlocked: Achievement[] = [];

    for (const achievement of ACHIEVEMENT_LIST) {
      if (unlockedIds.includes(achievement.id)) continue;

      const getter = this.listeners.get(achievement.condition.type);
      if (!getter) continue;

      let current = getter(state);
      let target: number;

      if ('target' in achievement.condition) {
        target = achievement.condition.target;
        if (current >= target) {
          newlyUnlocked.push(achievement);
        }
      }
    }

    return newlyUnlocked;
  }
}
```

```typescript
// services/welfareService.ts

interface SignInState {
  consecutiveDays: number;    // 连续签到天数
  totalDays: number;          // 累计签到天数
  lastSignInDate: string;     // 最后签到日期 YYYY-MM-DD
  currentMonthSigned: number[]; // 本月已签到日期列表
}

const SIGN_IN_REWARDS: Record<number, { silver: number; luckyCharm: number }> = {
  1:  { silver: 100,  luckyCharm: 1 },
  2:  { silver: 200,  luckyCharm: 1 },
  3:  { silver: 300,  luckyCharm: 2 },
  4:  { silver: 400,  luckyCharm: 2 },
  5:  { silver: 500,  luckyCharm: 3 },
  6:  { silver: 600,  luckyCharm: 3 },
  7:  { silver: 1000, luckyCharm: 5 }, // 周奖励
};

function checkIn(state: SignInState): { success: boolean; reward: SignInReward } {
  const today = getTodayStr();

  if (state.lastSignInDate === today) {
    return { success: false, reward: null };
  }

  const yesterday = getYesterdayStr();
  if (state.lastSignInDate === yesterday) {
    state.consecutiveDays += 1;
  } else {
    state.consecutiveDays = 1; // 断签重置
  }

  state.lastSignInDate = today;
  state.totalDays += 1;
  state.currentMonthSigned.push(new Date().getDate());

  const dayIndex = ((state.consecutiveDays - 1) % 7) + 1;
  const baseReward = SIGN_IN_REWARDS[dayIndex];

  return { success: true, reward: baseReward };
}
```

#### 3.5.2 家族金猪（定期储蓄玩法）

```typescript
interface GoldenPigState {
  depositedAmount: number;     // 当前已存入
  depositDays: number;         // 已连续存入天数
  lastDepositDate: string;
  interestRate: number;        // 利息率（随连续天数提升）
  maxDays: 7;                  // 满7天可取
}

const INTEREST_RATES = [0.02, 0.04, 0.06, 0.08, 0.12, 0.16, 0.25]; // 第1-7天利率

function depositToPig(state: GoldenPigState, amount: number): void {
  state.depositedAmount += amount;
  state.depositDays = Math.min(state.depositDays + 1, 7);
  state.lastDepositDate = getTodayStr();

  // 利率随连续天数提升
  state.interestRate = INTEREST_RATES[state.depositDays - 1];
}

function withdrawPig(state: GoldenPigState): number {
  if (state.depositDays < 7) return 0;

  const total = state.depositedAmount * (1 + state.interestRate);
  // 重置
  state.depositedAmount = 0;
  state.depositDays = 0;
  state.interestRate = INTEREST_RATES[0];

  return Math.floor(total);
}
```
